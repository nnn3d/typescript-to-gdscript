## TypeScript types in, readable GDScript out

```ts scene title="TypeScript"
export type Element = 'fire' | 'ice' | 'poison';

export interface Hit {
  damage: int;
  element: Element;
}

export class Health extends Node {
  health_changed = gd.signal<[new_health: int, element: Element]>();

  @exports weakness: Element = 'fire';
  @exports health: int = 100;
  @onready label: Label = this.get_node('%Label');

  take_hit(hit: Hit) {
    let damage = hit.damage;
    // A weak spot takes double damage.
    if (hit.element == this.weakness) {
      damage *= 2;
    }
    this.health = maxi(this.health - damage, 0);
    this.label.text = `HP: ${this.health}`;
    this.health_changed.emit(this.health, hit.element);
  }
}
```

```gdscript title="GDScript"
class_name Health
extends Node

signal health_changed(new_health: int, element: String)
@export
var weakness: String = "fire"
@export
var health: int = 100
@onready
var label: Label = self.get_node("%Label")

func take_hit(hit):
	var damage = hit.get("damage")
	# A weak spot takes double damage.
	if hit.get("element") == self.weakness:
		damage *= 2
	self.health = maxi(self.health - damage, 0)
	self.label.text = "HP: " + str(self.health)
	self.health_changed.emit(self.health, hit.get("element"))
```
